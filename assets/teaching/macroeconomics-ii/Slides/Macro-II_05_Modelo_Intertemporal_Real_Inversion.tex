\section{Punto de partida}


\begin{frame}{Objetivos}
  \begin{itemize}
    \setlength{\itemsep}{0.55em}
    \item Integrar las decisiones de consumo, ahorro, trabajo e inversión en un modelo dinámico de equilibrio general.
    \item Construir un modelo que sirva como base para analizar cómo choques macroeconómicos afectan a la economía, y para evaluar el papel de la política macroeconómica.
    \item \textbf{Modelo intertemporal del sector real de la economía}: mostrar cómo se determinan la producción real agregada, el consumo real, la inversión real, el empleo, el salario real y la tasa de interés real en la macroeconomía.
    \item Para predecir variables nominales, necesitamos añadir dinero al modelo intertemporal del sector real (próximas sesiones).
    \item El aspecto dinámico del modelo consiste en que hogares y empresas toman \textbf{decisiones intertemporales}: \textit{trade-offs} entre el presente y el futuro.
  \end{itemize}
\end{frame}


\begin{frame}{Decisión intertemporal de la empresa: Inversión}
  \small
  \begin{itemize}
    \item Inversión: gasto en maquinaria, equipo, plantas productivas, etc.
    \item Los \textbf{bienes de inversión} son producidos hoy para su uso futuro en la producción de bienes y servicios.
    \item La capacidad productiva utilizada para producir bienes de inversión podría utilizarse para producir bienes de consumo presente; por otro lado, los bienes de inversión \textbf{incrementan la capacidad productiva futura: el stock de capital futuro}.
    \item Por lo tanto, las empresas hacen un \textbf{\textit{trade-off} entre beneficios presentes y mayores beneficios futuros}.
    \item \textbf{Determinantes de la inversión}: el nivel presente de stock de capital, mayor productividad total de los factores esperada y la tasa de interés real.
    \item Un determinante adicional: percepción de \textbf{riesgo crediticio}. Si los prestamistas (bancos e instituciones financieras) perciben el crédito como más riesgoso en general, las empresas tendrán mayor dificultad en obtener crédito para financiar sus proyectos de inversión.
  \end{itemize}
\end{frame}




\begin{frame}{Setup del modelo}
  \begin{columns}[T,onlytextwidth]
    \begin{column}{0.49\textwidth}
      \textbf{Agentes}
      \begin{itemize}
        \item Hogar representativo.
        \item Empresa representativa.
        \item Gobierno.
      \end{itemize}
    \end{column}
    \begin{column}{0.49\textwidth}
      \textbf{Mercados}
      \begin{itemize}
        \item Laboral, con precio \(w\).
        \item Bienes y servicios, cuyo precio se normaliza a uno.
        \item Crédito, con rendimiento \(r\).
      \end{itemize}
    \end{column}
  \end{columns}
  \vspace{0.4cm}
  \begin{itemize}
    \item Especificamos el modelo en términos de curvas de oferta y demanda.
    \item Podrémos capturar la conducta esencial en el modelo al examinar la \textbf{interacción entre el hogar representativo, la empresa representativa y el gobierno en dos mercados: el mercado laboral en el periodo presente y el merado de bienes y servicios en el periodo presente}.
  \end{itemize}
  \notafuentes{Williamson (2018), pp. 401--402.}
\end{frame}

\begin{frame}{Secuencia económica del modelo}
  \begin{enumerate}
    \setlength{\itemsep}{0.58em}
    \item El hogar elige consumo y ocio en ambos periodos para maximizar su utilidad.
    \item La empresa contrata trabajo, produce el bien de consumo y elige inversión presente.
    \item La inversión determina el capital disponible en el periodo futuro: $K'=(1-\delta)K+I$.
    \item Gobierno, hogar y empresa demandan el bien presente; gobierno y hogar demandan el bien futuro.
    \item \(w, w'\) y \(r\) se ajustan para hacer compatibles todas las decisiones.
  \end{enumerate}
\end{frame}

\section{Hogar representativo}

\begin{frame}{El hogar representativo}
  El hogar dispone de \(h\) unidades de tiempo en cada periodo y elige
  \[
    (C,\ell,C',\ell',S^p).
  \]
  \begin{itemize}
    \setlength{\itemsep}{0.45em}
    \item Oferta laboral: \(N^s=h-\ell\) y \(N^{s\prime}=h-\ell'\).
    \item Ingreso laboral: \(w(h-\ell)\) y \(w'(h-\ell')\).
    \item Recibe beneficios \((\pi,\pi')\) porque es propietario de la empresa.
    \item Paga impuestos de suma fija \((T,T')\).
    \item Toma \((w,w',r)\) como dados.
  \end{itemize}
  \notafuentes{Williamson (2018), pp. 401--402.}
\end{frame}

\begin{frame}{Restricciones presupuestarias por periodo}
  En el presente,
  \[
    C+S^p=w(h-\ell)+\pi-T.
  \]
  En el futuro, el hogar consume todos sus recursos:
  \[
    C'=w'(h-\ell')+\pi'-T'+(1+r)S^p.
  \]
  \begin{itemize}
    \item \(S^p>0\): el hogar es ahorrador.
    \item \(S^p<0\): el hogar es deudor.
  \end{itemize}
  \notafuentes{Williamson (2018), ecuaciones 11-1 y 11-2.}
\end{frame}

\begin{frame}{Restricción presupuestaria intertemporal}
  Eliminando \(S^p\), obtenemos
  \[
    \boxed{
    C+\frac{C'}{1+r}
    =w(h-\ell)+\pi-T
    +\frac{w'(h-\ell')+\pi'-T'}{1+r}
    }.
  \]
  \begin{itemize}
    \setlength{\itemsep}{0.48em}
    \item El lado izquierdo es el valor presente del consumo.
    \item El lado derecho es la \textit{riqueza intertemporal disponible}.
    \item A diferencia del modelo del Cap. 9 (equilibrio parcial), el ingreso disponible por periodo es \textbf{endógeno} (equilibrio general).
  \end{itemize}
  \notafuentes{Williamson (2018), ecuación 11-3.}
\end{frame}

\begin{frame}{Problema del hogar}
  El hogar resuelve
  \[
    \max_{C,C',\ell,\ell'} U(C,\ell,C',\ell') = u(C, \ell) + \beta u(C', \ell')
  \]
  sujeto a
  \[
    C+\frac{C'}{1+r}
    \leq w(h-\ell)+\pi-T
    +\frac{w'(h-\ell')+\pi'-T'}{1+r}.
  \]
  Suponemos que \(U\) es creciente en consumo y ocio, estrictamente cuasicóncava y diferenciable, de modo que la solución interior es única.
  \notafuentes{Williamson (2018), pp. 402--403.}
\end{frame}

\begin{frame}{CPOs}
  \begin{align*}
    &C: \ u_C - \lambda = 0                         \tag{1} \\[4pt]
    &C': \  \beta u_{C'} - \frac{\lambda}{1+r} = 0    \tag{2}  \\[4pt]
    &\ell: \  u_\ell - w\lambda = 0                      \tag{3}  \\[4pt]
    &\ell': \  \beta u_{\ell'} - \frac{w'\lambda}{1+r} = 0 \tag{4} \\[4pt]
    &\lambda: \ w(h-\ell)+\pi-T +\frac{w'(h-\ell')+\pi'-T'}{1+r} - C - \frac{C'}{1+r} = 0 \tag{5}
  \end{align*}
\end{frame}

\begin{frame}{Tres condiciones de optimalidad}
  Sustituyendo (1) en (2):
  $$
  \boxed{u_C = \beta (1+r) u_{C'}} \qquad \text{\textbf{Ecuación de Euler}}
  $$
  Sustituyendo (1) en (3):
  $$
  \boxed{\mathrm{MRS}_{\ell,C}=w} \qquad \text{\textbf{Elección de ocio presente}}
  $$
  Sustituyendo (2) en (4):
  $$
  \boxed{\mathrm{MRS}_{\ell',C'}=w'} \qquad \text{\textbf{Elección de ocio futura}}
  $$
  \begin{itemize}
    \item El salario real presente (futuro) es el costo de oportunidad del ocio presente (futuro).
  \end{itemize}
\end{frame}

\begin{frame}{Oferta laboral en el tiempo}
  \textbf{¿En qué periodo el hogar elegirá trabajar más?}
  
  \vspace{.2cm}

  Sustituyendo (3) en (4):
  \begin{align*}
    \frac{u_\ell}{u_{\ell'}} = \beta(1+r) \frac{w}{w'}
  \end{align*}
  \begin{itemize}
    \small
    \item El ocio será relativamente más alto en el periodo en que el salario sea relativamente más bajo: \textbf{en periodos de mayor nivel salarial, habrá mayor oferta laboral}.
    \item Además, aumentos en $r$ incrementan la oferta laboral: mayor retorno al ahorro $\implies$ mayores incentivos a obtener ingresos laborales $\implies$ mayor ocio y consumo en el futuro.
    \item Esta respuesta de la oferta laboral, motivada por esta \textbf{sustitución intertemporal de trabajo}, está en el corazón de las fluctuaciones de empleo en este modelo.
  \end{itemize}
\end{frame}


\begin{frame}{Determinantes de la oferta laboral presente}
  Las decisiones óptimas generan una \textbf{función de oferta laboral}
  \[
    N^s=N^s(w, r; \omega),
  \]
  donde \(\omega\) denomina la riqueza intertemporal.
  \begin{itemize}
    \setlength{\itemsep}{0.5em}
    \item \(N^s_w>0\): un salario real mayor encarece el costo de oportunidad del ocio (suponemos que \textbf{el efecto sustitución ante incrementos salariales domina al efecto ingreso}).
    \item \(N^s_r>0\): una tasa mayor encarece el ocio presente respecto del ocio futuro (suponemos que \textbf{el efecto sustitución ante incrementos en $r$ domina al efecto ingreso}).
    \item \(N^s_\omega<0\): aumentar la riqueza intertemporal genera un \textbf{efecto ingreso}.
  \end{itemize}
\end{frame}

\begin{frame}{Respuesta de la oferta laboral presente: $w$}
    \begin{columns}[T,onlytextwidth]
    \begin{column}{0.4\textwidth}
      \justifying
      \small
      \begin{itemize}
        \setlength{\itemsep}{0.48em}
        \item Si \(w\uparrow\), el costo de oportunidad del ocio aumenta.
        \item El hogar \textbf{sustituye ocio por consumo}, suponiendo que domina el efecto sustitución del salario.
        \item Por lo tanto, la curva de oferta laboral tiene pendiente positiva.
      \end{itemize}
    \end{column}
    \begin{column}{0.55\textwidth}
      \centering
      \figura[5.3cm]{fig_11_1_w.png}
      \fuentefigura{Williamson (2018), figura 11.1.}
    \end{column}
  \end{columns}
\end{frame}


\begin{frame}[t]{Respuesta de la oferta laboral presente: $r$}
  \begin{columns}[T,onlytextwidth]
    \begin{column}{0.41\textwidth}
      \small
      \begin{itemize}
        \setlength{\itemsep}{0.48em}
        \item Si \(r\uparrow\), trabajar y ahorrar hoy permite adquirir más consumo y ocio mañana.
        \item El hogar \textbf{sustituye ocio presente por ocio futuro}.
        \item Para todo \(w\), \(N^s\) aumenta.
      \end{itemize}
    \end{column}
    \begin{column}{0.55\textwidth}
      \centering
      \figura[5.3cm]{fig_11_2_oferta_labor_tasa.png}
      \fuentefigura{Williamson (2018), figura 11.2.}
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}{Respuesta de la oferta laboral presente: $\omega$}
    \begin{columns}[T,onlytextwidth]
    \begin{column}{0.41\textwidth}
      \small
      \begin{itemize}
        \setlength{\itemsep}{0.48em}
        \item Si la riqueza intertemporal aumenta \(\omega\uparrow\), se genera un efecto ingreso.
        \item El hogar demanda mayor consumo y ocio (bienes normales).
        \item Para todo \(w\), \(N^s\) cae.
      \end{itemize}
    \end{column}
    \begin{column}{0.55\textwidth}
      \centering
      \figura[5.3cm]{fig_11_3_oferta_laboral_we.png}
      \fuentefigura{Williamson (2018), figura 11.3.}
    \end{column}
  \end{columns}
\end{frame}



\begin{frame}[t]{Determinantes de la demanda de consumo presente}
  \begin{columns}[T,onlytextwidth]
    \begin{column}{0.45\textwidth}
      \small
      \justifying
      \begin{itemize}
        \item La ecuación de Euler genera una \textbf{función de demanda de consumo} presente
        \[
          C^d=C^d(r; Y, \omega).
        \]
        donde $Y$ es el ingreso corriente.
        \item La propensión marginal a consumir satisface
        \[
          0<C_Y^d<1.
        \]
        Parte de un aumento del ingreso presente se ahorra (por suavización del consumo).
      \end{itemize}
    \end{column}
    \begin{column}{0.49\textwidth}
      \centering
      \figura[5.3cm]{fig_11_4_consumo_ingreso.png}
      \fuentefigura{Williamson (2018), figura 11.4.}
    \end{column}
  \end{columns}
  \notafuentes{Williamson (2018), figura 11.4 y pp. 406--407.}
\end{frame}

\begin{frame}{Desplazamientos de la demanda de consumo presente}
  Manteniendo constante el ingreso presente \(Y\):
  \begin{itemize}
    \setlength{\itemsep}{0.62em}
    \item Si \(r\uparrow \Rightarrow C^d\downarrow\), suponiendo que domina el efecto sustitución intertemporal.
    \item Si \(Y'\uparrow \Rightarrow C^d\uparrow\), por suavización del consumo.
    \item Si \(\omega \uparrow \Rightarrow C^d\uparrow\), por suavización del consumo.
    \item Por HIP, el desplazamiento de la curva de demanda de consumo debe ser mayor cuando aumenta $\omega$ respecto a cuando aumentan $Y$ o $Y'$.
  \end{itemize}
\end{frame}

\section{Empresa representativa}


\begin{frame}{Tecnología en los dos periodos}
  La empresa produce utilizando capital y trabajo como insumos productivos:
  \[
    Y=zF(K,N),
    \qquad
    Y'=z'F(K',N').
  \]
  \begin{itemize}
    \setlength{\itemsep}{0.55em}
    \item \(z\) y \(z'\) son la productividad total de los factores presente y futura.
    \item \(K\) está predeterminado al inicio del periodo presente.
    \item \(K'\) es elegido indirectamente mediante la inversión actual.
    \item \(F\) tiene productos marginales positivos y decrecientes.
  \end{itemize}
\end{frame}

\begin{frame}{Acumulación de capital}
  La \textbf{ley de movimiento del capital} es
  \[
    \boxed{K'=(1-\delta)K+I},
    \qquad 0\leq\delta\leq1.
  \]
  \begin{itemize}
    \setlength{\itemsep}{0.58em}
    \item $\delta$ es la tasa de depreciación del capital.
    \item Una unidad de producto presente se transforma uno a uno en una unidad de capital futuro.
    \item \((1-\delta)K\) es el capital presente que sobrevive hasta el futuro.
    \item Al final del segundo periodo, el capital remanente se vende como bien de consumo.
  \end{itemize}
  \notafuentes{Williamson (2018), ecuación 11-9.}
\end{frame}


\begin{frame}{Beneficios presentes y futuros}
  Los beneficios son
  \[
    \pi=Y-wN-I,
  \]
  \[
    \pi'=Y'-w'N'+(1-\delta)K'.
  \]
  La empresa maximiza el valor presente de sus beneficios:
  \[
    \boxed{V=\pi+\frac{\pi'}{1+r}}.
  \]
  \begin{itemize}
    \item Invertir reduce los dividendos presentes en una unidad por unidad invertida.
    \item El beneficio aparece mañana mediante mayor producción ($Y'$) y mayor valor terminal (i.e. mayor $(1-\delta)K'$).
  \end{itemize}
  \notafuentes{Williamson (2018), ecuaciones 11-10--11-12.}
\end{frame}

\begin{frame}{Problema de la empresa}
  Tomando \((w,w',r)\) como dados, la empresa resuelve
  \[
    \max_{N,N',I}\;
    V = zF(K,N)-wN-I
    +\frac{z'F((1-\delta)K+I,N')-w'N'+(1-\delta)K'}{1+r}.
  \]
  CPO:
  \begin{align*}
    &N: \ zF_N - w = 0 \tag{1} \\[4pt]
    &N': \ \frac{z'F_{N'} - w'}{1+r} = 0 \tag{2} \\[4pt]
    &I': \ -1 + \frac{z'F_{K'} +(1-\delta)}{1+r} = 0 \tag{3}
  \end{align*}
\end{frame}



\begin{frame}{Condiciones de optimalidad de la demanda laboral}
  De (1) y (2), las relgas de decisión de demanda laboral son:
  \[
    \boxed{\mathrm{MPN}=w},
    \qquad
    \boxed{\mathrm{MPN}'=w'}.
  \]
\end{frame}


\begin{frame}[t]{Determinantes de la demanda laboral presente}
  \begin{columns}[T,onlytextwidth]
    \begin{column}{0.45\textwidth}
      \justifying
      \small
      \begin{itemize}
        \item La empresa contrata trabajo hasta que
        $$
          zF_N(K,N^d)=w.
        $$
        \item Esta regla óptima de decisión genera una función de demanda laboral
        $$
        N^d = N^d(w; z, K)
        $$
      \end{itemize}
    \end{column}
    \begin{column}{0.55\textwidth}
      \centering
      \figura[5.3cm]{fig_11_7_demanda_labor.png}
      \fuentefigura{Williamson (2018), figura 11.7.}
    \end{column}
  \end{columns}
  \notafuentes{Williamson (2018), figura 11.7 y pp. 410--411.}
\end{frame}


\begin{frame}{Respuesta de la demanda laboral a $w$}
  \small 
  Para $z$ y $K$ dados:
  \begin{itemize}
    \item Partimos de un equilibrio inicial $zF_N(K,N_1) = w_1$.
    \item Si $w_2 > w_1$, entonces al nivel de empleo inicial $N_1$:
    $$
      zF_N(K,N_1) <  w_2
    $$
    \item Como \(F_{NN}<0\), la empresa responde a un salario mayor contratando menos trabajadores.
    \item Reduce su demanda laboral hasta el nivel $N_2<N_1$ tal que: 
    $$
      zF_N(K,N_2) =  w_2
    $$
    \item Formalmente, diferenciando \(zF_N(K,N^d)=w\) respecto de \(w\),
        \[
          zF_{NN}(K,N^d)N^d_w=1
          \implies 
          \boxed{
          N^d_w
          =
          \frac{1}{zF_{NN}(K,N^d)}<0
          }
        \]
  \end{itemize}
\end{frame}




\begin{frame}{Desplazamientos de la demanda laboral}
  \small
  Para un nivel salarial $w$ dado:
  \begin{itemize}
    \setlength{\itemsep}{0.5em}
    \item Si $z_2>z_1$, entonces aumenta la $\mathrm{MPN}$ para todo $N$, de modo que, al salario vigente y a un nivel de empleo inicial $N_1$:
    $$
    z_2F_N(K, N_1) > w
    $$
    Esto genera incentivos a contratar más trabajadores, desplazando $N^d$ hacia la derecha hasta el nivel de empleo $N_2>N_1$ tal que: $z_2F_N(K, N_2) = w$.
    \item Si $K_2>K_1$, entonces aumenta la $\mathrm{MPN}$ para todo $N$, de modo que, al salario vigente y a un nivel de empleo inicial $N_1$:
    $$
    zF_N(K_2, N_1) > w
    $$
    Esto genera incentivos a contratar más trabajadores, desplazando $N^d$ hacia la derecha hasta el nivel de empleo $N_2>N_1$ tal que: $zF_N(K_2, N_2) = w$.
  \end{itemize}
\end{frame}

\begin{frame}[label=reglainversion]{Regla óptima de inversión}
  La condición (3) indica que, en el óptimo, el costo marginal en valor presente de la inversión, que es igual a 1, es igual al beneficio marginal en valor presente de la inversión:
  \[
  \boxed{
    \frac{\mathrm{MPK'} + 1-\delta}{1+r} = 1
  }
  \]
  Reordenando la condición de optimalidad, obtenemos la \textbf{regla óptima de inversión}
  \[
  \boxed{
    \mathrm{MPK'} -\delta = r
  }
  \] 
  La empresa invierte hasta el punto en que el \textbf{producto marginal neto del capital}, $\mathrm{MPK'} - \delta$, es igual a la tasa de interés real.

  \hyperlink{apendice-inversion}{
    \beamergotobutton{Ver Costo Marginal y Beneficio Marginal de la Inversión}
  }
\end{frame}


\begin{frame}{Interpretación de la regla óptima de inversión}
    \begin{itemize}
    \item \(\mathrm{MPK}'-\delta\) es el rendimiento marginal neto del capital.
    \item \(r\) es el costo de oportunidad (el rendimiento disponible en bonos) y el costo efectivo de los proyectos de inversión.
    \item Si \(\mathrm{MPK}'-\delta>r\), la empresa incrementa \(I\).
    \item Si \(\mathrm{MPK}'-\delta<r\), la empresa reduce \(I\).
  \end{itemize}
\end{frame}

\begin{frame}[t]{Plan óptimo de inversión}
  \begin{columns}[T,onlytextwidth]
    \begin{column}{0.49\textwidth}
      \justifying
      \small
      \begin{itemize}
        \item La condición de optimalidad genera una \textbf{función de demanda de inversión}:
        \[
          I^d=I^d(r;K,z',\delta)
        \]
        \item La Figura 11.9 grafica el \textbf{plan óptimo de inversión}: para cada $r$, el nivel $I^d$ tal que
        $$
        \mathrm{MPK'}(I^d) - \delta = r
        $$
        \item Como \(F_{KK}<0\), diferenciando respecto a $r$:
        $$
        z'F_{K'K'} I^d_r = 1 \implies I^d_r = \frac{1}{z'F_{K'K'}(K',N')} < 0
        $$
      \end{itemize}

\end{frame}

\begin{frame}{¿Por qué la inversión es tan volátil?}
  \begin{itemize}
    \setlength{\itemsep}{0.63em}
    \item El consumo responde a incentivos de suavización intertemporal.
    \item La inversión responde al rendimiento marginal neto esperado de proyectos de inversión.
    \item Fluctuaciones relativamente pequeñas en \(r\), \(z'\), \(K\) o el riesgo de crédito pueden generar fluctuaciones grandes en \(I\).
    \item La inversión absorbe una parte importante del ajuste ante choques agregados.
  \end{itemize}
  \notafuentes{Williamson (2018), pp. 415--416.}
\end{frame}

\begin{frame}{Información asimétrica y prima de incumplimiento}
  \small
  \begin{itemize}
    \item Información asimétrica: las empresas conocen su calidad crediticia, pero los prestamistas (bancos e instituciones financieras) no.
    \item Como los prestamistas no pueden distinguir entre empresas buenas y empresas malas, cobran una prima por riesgo crediticio sobre la tasa de interés real vigente:
    \[
      r^L=r+x,
    \]
    donde \(x>0\) es la \textbf{prima de incumplimiento}. 
    
    \item La regla óptima de inversión de la empresa entonces es
    \[
      \boxed{\mathrm{MPK}'-\delta=r+x}
      \qquad\Longleftrightarrow\qquad
      \mathrm{MPK}'-\delta-x=r.
    \]
    \item Una mayor incertidumbre crediticia eleva el costo de financiamiento a través de la prima de riesgo crediticio, lo que reduce la inversión para cualquier tasa de interés real libre de riesgo \(r\).
  \end{itemize}
\end{frame}

\begin{frame}[t]{Una prima mayor reduce la inversión}
  \begin{columns}[T,onlytextwidth]
    \begin{column}{0.40\textwidth}
      \justifying
      \small
      \begin{itemize}
        \item \(x_2>x_1\) aumenta el costo de financiamiento.
        \item La inversión se reduce para cualquier tasa de interés libre de riesgo $r$.
        \item Incluso empresas solventes invierten menos si el mercado no puede identificarlas.
      \end{itemize}
    \end{column}
    \begin{column}{0.56\textwidth}
      \centering
      \figura[5.3cm]{fig_11_11_prima_incumplimiento.png}
      \fuentefigura{Williamson (2018), figura 11.11.}
    \end{column}
  \end{columns}
\end{frame}

\section{Equilibrio general}

\begin{frame}{Gobierno}
  El gobierno fija exógenamente \((G,G')\), cobra impuestos de suma fija \((T,T')\) y puede emitir deuda. Su restricción intertemporal es
  \[
    \boxed{G+\frac{G'}{1+r}=T+\frac{T'}{1+r}}.
  \]
  \begin{itemize}
    \setlength{\itemsep}{0.55em}
    \item El valor presente del gasto debe financiarse con el valor presente de impuestos.
    \item Un aumento de \(G\) o \(G'\) reduce la riqueza intertemporal del hogar representativo.
    \item Equivalencia ricardiana: el momento de los impuestos no altera la riqueza intertemporal del hogar.
  \end{itemize}
  \notafuentes{Williamson (2018), ecuación 11-18.}
\end{frame}

\begin{frame}{Definición de equilibrio competitivo}
  \small
  \textit{Definición.} Dadas las variables exógenas \((K,z,z',G,G')\), un \textit{equilibrio competitivo} consiste en una asignación $(C^*, \ell^*, C'^{*}, \ell'^{*}, I^*, K'^{*}, N^*, N'^{*})$ y un vector de precios $(w^*, w'^{*}, r^*)$ tales que:
  \begin{enumerate}
    \setlength{\itemsep}{0.48em}
    \item el hogar resuelve su problema intertemporal de maximización de utilidad, tomando precios como dados;
    \item la empresa maximiza el valor presente de sus beneficios, tomando precios como dados;
    \item el gobierno satisface su restricción presupuestaria intertemporal;
    \item los mercados de trabajo, de crédito y de bienes se vacían en ambos periodos.
  \end{enumerate}
\end{frame}

\begin{frame}{Bienestar en equilibrio general dinámico}
  \begin{itemize}
    \item Ya hemos caracterizado el equilibrio en este \textbf{modelo de equilibrio general dinámico del sector real}.
    \item En equilibrio general estático, vimos que la asignación de un equilibrio competitivo es óptima en el sentido de Pareto.
    \item ¿Podemos extender este resultado al modelo de equilibrio general en dos periodos?
  \end{itemize}
\end{frame}

\begin{frame}{Problema del planificador social}
  Sabemos que la solución del problema del planificador social es una asignación Pareto óptima.

  \vspace{0.4cm}

  El planificador social elige un asignación que maximice el bienestar del hogar sujeto a las posibilidades tecnológicas:
  \small
  \begin{align*}
    \max_{C,C',\ell,\ell'} U(C,\ell,C',\ell') &= u(C, \ell) + \beta u(C', \ell')\\
    &\text{\textit{s.a.}} \\[4pt]
    K' &\leq (1-\delta)K + F(K,N) - C \\[4pt]
    C' &\leq F(K',N') \\[4pt]
    N &\leq 1 - \ell, \quad N' \leq 1 - \ell' \\[4pt]
    K &\text{ dado }
  \end{align*}
\end{frame}


\begin{frame}{Primer teorema del bienestar}
  \begin{proposition}
    Si la asignación $(C^*, \ell^*, C'^{*}, \ell'^{*}, I^*, K'^{*}, N^*, N'^{*})$ es parte de un equilibrio competitivo, entonces resuelve el problema del planificador social.
  \end{proposition}
  \begin{proof}
    Ver Kurlat (2020, pp. 170-171).
  \end{proof}
\end{frame}




\begin{frame}{Análisis gráfico del equilibrio}
  \begin{itemize}
    \item Para estudiar la respuesta de equilibrio de esta economía ante distintos choques macroeconómicos, utilizaremos el \textbf{análisis gráfico}.
    
    \item Nos concentraremos en el equilibrio de los mercados de trabajo y de bienes en el \textbf{periodo presente}.
    
    \item Sin embargo, no debemos perder de vista el \textbf{carácter dinámico del modelo}: las decisiones presentes dependen de las expectativas que los agentes tienen sobre las condiciones económicas futuras.
  \end{itemize}
\end{frame}


\begin{frame}{Curvas de oferta y demanda de producto}
  \begin{itemize}
    \setlength{\itemsep}{0.65em}
    \item A partir del equilibrio en el mercado laboral, construiremos una \textbf{curva de oferta de producto}, \(Y^s(r)\), que describe \textit{cómo se relaciona la cantidad ofrecida de bienes con la tasa de interés real}.

    \item A partir del equilibrio en el mercado de bienes, construiremos una \textbf{curva de demanda de producto}, \(Y^d(r)\), que describe \textit{cómo se relaciona la cantidad demandada de bienes con la tasa de interés real}.

    \item Finalmente, la intersección de ambas curvas determinará el producto y la tasa de interés real de equilibrio:
    \[
      \boxed{Y^s(r^*)=Y^d(r^*)=Y^*}.
    \]
  \end{itemize}
\end{frame}



\begin{frame}{Construcción de la curva de oferta de producto (I)}
  \begin{itemize}
    \item Determinación de la producción agregada.
      \begin{enumerate}
        \item \(N^s(w,r)\) y \(N^d(w)\) determinan \((w^*,N^*)\) para un valor dado de \(r\).
        \vspace{0.2cm}
        \item La función de producción determina el nivel de producción ofrecido: $Y^*=zF(K,N^*)$.
      \end{enumerate}
    \item Curva de oferta de producto:
    \begin{enumerate}
      \item Si $r \uparrow$, entonces: por sustitución intertemporal de la oferta laboral, $N^s(w,r)$ se desplaza hacia la derecha $\implies$ $w^* \downarrow$ y $N^* \uparrow$ $\implies$ aumenta la producción agregada: $Y^* \uparrow$. 
      \vspace{0.2cm}
      \item Por lo tanto, en el plano $(Y,r)$, la \textbf{curva de oferta de producto} tiene pendiente positiva.
    \end{enumerate}
  \end{itemize}
\end{frame}

\begin{frame}{Desplazamientos de la curva de oferta de producto \(Y^s(r)\)}
  \(Y^s(r)\) se desplaza a la derecha cuando:
  \begin{itemize}
    \setlength{\itemsep}{0.58em}
    \item \(z\uparrow\): $\mathrm{MPN} \uparrow \implies N^d \uparrow \implies w^* \uparrow, \ N^* \uparrow \implies  Y^* \uparrow$
    \item \(K\uparrow\): $\mathrm{MPN} \uparrow \implies N^d \uparrow \implies w^* \uparrow, \ N^* \uparrow \implies  Y^* \uparrow$
    \item \(G\uparrow\) o \(G'\uparrow\): $\omega \downarrow \implies N^s \uparrow \implies w^* \downarrow, \ N^* \uparrow \implies Y^* \uparrow$
  \end{itemize}
  \begin{alertblock}{Movimiento vs desplazamiento}
    Un cambio en \(r\) genera un movimiento sobre \(Y^s\); cambios en \(z\), \(K\) o la riqueza intertemporal desplazan la curva.
  \end{alertblock}
\end{frame}


\begin{frame}{Demanda agregada de bienes}
  \small
  \begin{itemize}
    \setlength{\itemsep}{0.55em}

    \item Para una tasa de interés real \(r\) y un nivel dado de ingreso corriente \(Y\), el gasto agregado deseado en bienes es
    \[
      C^d(Y,r)+I^d(r)+G.
    \]
    
    \item Esta expresión indica cuánto desean gastar los hogares, las empresas y el gobierno para cada nivel de ingreso y tasa de interés.

    \item Por contabilidad nacional, la producción agregada $zF(K,N)$ es igual al ingreso agregado $Y$. 

    \item El mercado de bienes se encuentra en equilibrio cuando la producción es igual al gasto agregado deseado:
    \[
      \boxed{Y = zF(K,N) = C^d(Y,r)+I^d(r)+G}.
    \]
  \end{itemize}
\end{frame}

\begin{frame}{Desplazamientos de la demanda de producto}

  \begin{itemize}

    \item \(G\uparrow\): aumenta directamente la demanda agregada de bienes $Y^d \uparrow$.
    \item \(T + \frac{T'}{1+r}\) disminuye: \(\omega \uparrow \implies C^d \uparrow \implies Y^d \uparrow\).
    \item \(Y\uparrow\) o \(Y'\uparrow\) o ambas: $C^d(r) \uparrow \implies Y^d \uparrow$.
    \item \(z'\uparrow\): $\mathrm{MPK}' \uparrow \implies I^d(r) \uparrow \implies Y^d \uparrow$.
    \item \(K\downarrow\): $\mathrm{MPK}' \uparrow \implies I^d(r) \uparrow \implies Y^d \uparrow$.
  \end{itemize}

  Un cambio en \(r\) genera un movimiento sobre \(Y^d\); cambios en \(z'\), \(K\), $G$ o la riqueza intertemporal desplazan la curva.

\begin{frame}{Curvas de oferta y demanda agregada de producto}
  \begin{itemize}
    \item \textbf{Curva de oferta de producto, \(Y^s(r)\):} combinaciones de producto agregado real y tasa de interés real, $(Y,r)$, para las cuales el \textbf{mercado laboral se encuentra en equilibrio}.
    
    Para cada \(r\), el vaciado del mercado laboral determina \(w\), \(N(r)\) y, por lo tanto,
    \[
      Y^s(r)=zF\bigl(K,N(r)\bigr).
    \]
    \item \textbf{Curva de demanda de producto, \(Y^d(r)\):} combinaciones de producto agregado real y tasa de interés real, $(Y,r)$, para las cuales el \textbf{mercado de bienes se encuentra en equilibrio}:

      Y^d(r)
      =
      C^d\bigl(Y^d(r),r\bigr)+I^d(r)+G.
    \]
  \end{itemize}
\end{frame}



\begin{frame}{Orden práctico para resolver el modelo}
  \begin{enumerate}
    \setlength{\itemsep}{0.62em}
    \item Obtener \(N^s(w,r)\), \(C^d(Y,r)\), \(N^d(w)\) e \(I^d(r)\).
    \item Imponer \(N^s=N^d\) para construir \(Y^s(r)\).
    \item Imponer \(Y=C^d+I^d+G\) para construir \(Y^d(r)\).
    \item Resolver \(Y^s(r^*)=Y^d(r^*)\).
    \item Obtener \(Y^*,w^*,N^*,C^*,I^*\) y \(K^{\prime *}\).
  \end{enumerate}
\end{frame}



\section{Choques exógenos}



\begin{frame}{Cómo leer un choque en equilibrio general}
  Un choque modifica una o más reglas de decisión; esas modificaciones desplazan curvas y producen un nuevo vector de precios y asignaciones de equilibrio.
  \vspace{.2cm}
  \begin{enumerate}
    \setlength{\itemsep}{0.55em}
    \item Identificar la variable exógena que cambia y el mercado donde aparece directamente.
    \item Determinar qué reglas de decisión se modifican.
    \item Seguir la propagación del choque hacia el resto de las variables endógenas (apoyarse de muchos diagramas!).
    \item Traducir esos cambios en desplazamientos de \(Y^s(r)\), \(Y^d(r)\) o de las curvas del mercado laboral.
    \item Encontrar el nuevo \((Y^*,r^*)\) y el nuevo \((N^*,w^*)\).
  \end{enumerate}
\end{frame}

\begin{frame}{Aumento temporal del gasto público}
  Suponga que \(G\) aumenta en el periodo presente, mientras \(G'\) permanece constante.
  \begin{itemize}
    \setlength{\itemsep}{0.48em}
    \item El choque se origina en el mercado de bienes, desplazando $Y^d$ hacia la derecha (\textbf{choque de demanda}).
    \item Primero, analizamos el efecto de equilibrio parcial.
    \item A partir de aquí, analizamos los canales y mecanismos de transmisión en equilibrio general.
  \end{itemize}
\end{frame}

\begin{frame}{Desplazamiento inicial de la demanda de producto}
  Sea \(\Delta Y\) el desplazamiento horizontal inicial de \(Y^d\), manteniendo \(r\) fijo. Entonces,
  \[
    \Delta Y
    =
    (G_2 - G_1) -  C_\omega (G_2 - G_1) + C_\omega \Delta Y.
  \]
  donde $C_\omega$ es la propensión marginal a consumir respecto a la riqueza intertemporal.


  Por lo tanto,
  \[
    (1 - C_\omega) \Delta Y=(1 - C_\omega)(G_2 - G_1)
    \quad\Longrightarrow\quad
    \boxed{\Delta Y = G_2 - G_1}.
  \]


  A la tasa de interés inicial, el aumento del gasto público desplaza la curva de demanda de producto por el monto de \(G_2 - G_1\).

  \small

  $$
  \text{\textbf{Multiplicador fiscal}: } \quad 
  \boxed{
  \frac{\Delta Y}{\Delta G} = \frac{G_2 - G_1}{G_2 - G_1} = 1
  }
  $$
\end{frame}






\begin{frame}{Efectos de equilibrio general}
  \small
  \begin{itemize}
    \setlength{\itemsep}{0.45em}

    \item El aumento en \(G\) desplaza la curva de demanda de producto hacia la derecha. 

    \item A la tasa de interés inicial \(r_1\), el aumento en \(G\) reduce la riqueza intertemporal del hogar, de modo que aumenta su oferta laboral. Como resultado, el salario real disminuye, mientras que el empleo y la producción aumentan. Este efecto desplaza la curva de oferta de producto hacia la derecha.
    
    \item \textcolor{red}{\textbf{Suponemos que la respuesta de la oferta laboral es pequeña}}, de tal forma que \textcolor{red}{\textbf{el desplazamiento de la curva de oferta de producto es menor al de la curva de demanda}}.

    \item Por lo tanto, a la tasa \(r_1\) existe un exceso de demanda de bienes, lo que induce un aumento de la tasa de interés real de \(r_1\) a \(r_2\) (podríamos suponer que $Y^s\uparrow \ \geq Y^d\uparrow$, de modo que $r_2 \leq r_1$).

    \item Este aumento en \(r\) tiene dos efectos simultáneos:
    \begin{itemize}
      \item incrementa la oferta laboral y, por lo tanto, la producción;
      \item reduce el consumo y la inversión
    \end{itemize}
    Por lo tanto, la oferta y la demanda se ajustan \textbf{sobre la curva} hasta $Y_2$.
  \end{itemize}


\end{frame}

\begin{frame}{Multiplicador total del gasto público}
    Del vaciado del mercado de bienes,
    \[
      \Delta Y=\Delta C+\Delta I+\Delta G.
    \]
    Por lo tanto,
    \[
      \boxed{
        \frac{\Delta Y}{\Delta G}
        =
        1+\frac{\Delta C}{\Delta G}
        +\frac{\Delta I}{\Delta G}
        <1
      },
    \]
    porque \(\Delta C<0\) y \(\Delta I<0\).

    \vspace{.2cm}

    \cidehighlight{El producto aumenta, pero menos que proporcionalmente al impulso fiscal.}
\end{frame}

\begin{frame}{Destrucción del stock de capital}
  Una caída exógena de \(K\) opera por dos canales:
  \begin{enumerate}
    \setlength{\itemsep}{0.52em}
    \item \textbf{Oferta presente:} cae la productividad marginal del trabajo, de modo que \(N^d\) y \(Y^s\) se desplazan a la izquierda (\textbf{choque de oferta}).
    \item \textbf{Inversión:} aumenta el producto marginal futuro del capital, por lo que \(I^d\) y \(Y^d\) se desplazan a la derecha. La tasa real aumenta para reasignar bienes desde consumo hacia inversión (\textbf{choque de demanda}).
  \end{enumerate}
  El \textbf{efecto sobre producto y empleo son ambiguos}.
\end{frame}

\begin{frame}{Aumento de la productividad presente}
  Ante un aumento transitorio de \(z\):
  \begin{itemize}
    \setlength{\itemsep}{0.52em}
    \item La productividad marginal del trabajo aumenta: \(N^d\) y \(Y^s\) se desplazan a la derecha (\textbf{choque de oferta}).
    \item El exceso de demanda laboral genera un ajuste al alza en el salario real.
    \item El exceso de oferta de producto genera un ajuste a la baja en la tasa de interés, lo que incentiva al consumo presente y la demanda de inversión
    \item Por efecto de sustitución intertemporal, la oferta laboral se reduce ante la menor tasa de interés.
  \end{itemize}

  \textbf{El efecto neto sobre el nuevo empleo de equilibrio es ambiguo}.
\end{frame}

\begin{frame}{Expectativas sobre productividad futura}
  Suponga que aumenta \(z'\), sin cambio en \(z\).
  \begin{itemize}
    \setlength{\itemsep}{0.5em}
    \item El mayor producto marginal futuro del capital desplaza \(I^d(r)\) y \(Y^d(r)\) a la derecha (\textbf{choque de demanda}).
    \item La tasa real aumenta para equilibrar la mayor demanda de bienes.
    \item La mayor tasa real desplaza hacia la derecha la oferta laboral.
    \item El incremento en $z'$ genera que $Y^*$ y $Y'^*$ aumenten.
    \item Por lo anterior, \textcolor{red}{\textbf{el efecto sobre \(C\) es ambiguo, porque el ingreso permanente incrementa al mismo tiempo que el ahorro ofrece un mayor rendimiento}}.
  \end{itemize}
\end{frame}

\begin{frame}{Aumento de las fricciones financieras}
  Deudores y proyectos de inversión enfrentan un costo de financiamiento igual a:
  \[
    r+x
  \]
  Ahorradores enfrentan un rendimiento al ahorro igual a $r$.
  \begin{itemize}
    \setlength{\itemsep}{0.5em}
    \item Ante un aumento en $x$, \(I^d(r;x)\) se desplaza a la izquierda.
    \item La menor inversión reduce la demanda de producto \(Y^d(r)\).
    \item La mayor tasa de interés efectiva para los deudores les genera un efecto ingreso negativo que induce un aumento en su oferta laboral, de modo que $N^s(r)$ y \(Y^s(r)\) se desplazan a la derecha.
    \item La tasa de interés real libre de riesgo se ajusta a la baja para vaciar el mercado de bienes.
    \item Por lo tanto, \textbf{la economía termina en un equilibrio de menor tasa de interés real con mayores tasas de interés de mercado}.
  \end{itemize}
\end{frame}

\begin{frame}[label=apendice-inversion]{Costo marginal y beneficio marginal de la inversión}
  \footnotesize
  \begin{itemize}
    \item En el presente, el gasto en inversión es un costo para la empresa. Es decir, su \textbf{función de costos} es: $C(N) = wN + I$.
    \item El \textbf{costo marginal de la inversión} entonces es: $CMgI = 1$
    \item Como $\pi = Y - C(N)$, invertir reduce los dividendos presentes en una unidad por unidad invertida.
    \item El retorno de la inversión se obtiene en el futuro mediante mayor producción y mayor valor terminal. Es decir, el \textbf{beneficio marginal de la inversión} es:
    $$
    \frac{\partial \pi'}{ \partial I} = \frac{\partial}{\partial I} \left[Y' - w'N' + (1-\delta)K'\right] 
    = \frac{\partial Y'}{\partial I} + \frac{\partial \left[(1-\delta)K' \right]}{\partial I}
    = \mathrm{MPK'} + 1-\delta
    $$

    pues $Y' = z' F(K',N')$ y $K' = (1-\delta)K + I$.

    \item En el óptimo, el costo marginal de la inversión es igual al valor presente del beneficio marginal de la inversión:
    $$
      1 = \frac{\mathrm{MPK'} + 1-\delta}{1+r}
    $$
  \end{itemize}
